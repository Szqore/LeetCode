class Solution { //task 1510
    public boolean winnerSquareGame(int n) {
        // dp[i] показывает, выиграет ли текущий игрок, если осталось i камней
        boolean[] dp = new boolean[n + 1];
        
        // Заполняем массив dp от 1 до n
        for (int i = 1; i <= n; i++) {
            // Пробуем вычесть все возможные квадраты чисел: 1, 4, 9, 16...
            for (int k = 1; k * k <= i; k++) {
                // Если после нашего хода соперник оказывается в проигрышном состоянии,
                // то текущее состояние i — выигрышное для нас.
                if (!dp[i - k * k]) {
                    dp[i] = true;
                    break; // Оптимизация: нашли выигрышный ход, дальше можно не искать
                }
            }
        }
        
        return dp[n];
    }
}
