import java.util.Arrays;
///task 3348
public class Solution {
    public String smallestNumber(String num, long t) {
        // Шаг 1: Разложение t на простые множители
        long tempT = t;
        int req2 = 0, req3 = 0, req5 = 0, req7 = 0;
        while (tempT % 2 == 0) { req2++; tempT /= 2; }
        while (tempT % 3 == 0) { req3++; tempT /= 3; }
        while (tempT % 5 == 0) { req5++; tempT /= 5; }
        while (tempT % 7 == 0) { req7++; tempT /= 7; }
        
        // Если есть множители отличные от 2, 3, 5, 7 — составить число невозможно
        if (tempT > 1) return "-1";

        int n = num.length();
        
        // Префиксные массивы для подсчета накопленных нулей и множителей в num
        int[] count0 = new int[n + 1];
        int[] count2 = new int[n + 1];
        int[] count3 = new int[n + 1];
        int[] count5 = new int[n + 1];
        int[] count7 = new int[n + 1];

        for (int i = 0; i < n; i++) {
            char c = num.charAt(i);
            count0[i + 1] = count0[i] + (c == '0' ? 1 : 0);
            count2[i + 1] = count2[i] + getFactor(c, 2);
            count3[i + 1] = count3[i] + getFactor(c, 3);
            count5[i + 1] = count5[i] + getFactor(c, 5);
            count7[i + 1] = count7[i] + getFactor(c, 7);
        }

        // Проверяем, подходит ли исходное число num целиком
        if (count0[n] == 0 && count2[n] >= req2 && count3[n] >= req3 && count5[n] >= req5 && count7[n] >= req7) {
            return num;
        }

        // Шаг 2: Жадный поиск позиции для увеличения (справа налево)
        for (int i = n - 1; i >= 0; i--) {
            if (count0[i] > 0) continue; // Префикс с нулями использовать нельзя

            int c2 = count2[i], c3 = count3[i], c5 = count5[i], c7 = count7[i];
            char currentDigit = num.charAt(i);

            for (char d = (char)(currentDigit + 1); d <= '9'; d++) {
                int n2 = c2 + getFactor(d, 2);
                int n3 = c3 + getFactor(d, 3);
                int n5 = c5 + getFactor(d, 5);
                int n7 = c7 + getFactor(d, 7);

                int rem2 = Math.max(0, req2 - n2);
                int rem3 = Math.max(0, req3 - n3);
                int rem5 = Math.max(0, req5 - n5);
                int rem7 = Math.max(0, req7 - n7);

                int remLen = n - 1 - i;
                int minDigitsNeeded = getMinDigitsLen(rem2, rem3, rem5, rem7);

                if (minDigitsNeeded <= remLen) {
                    StringBuilder sb = new StringBuilder();
                    sb.append(num, 0, i);
                    sb.append(d);
                    
                    String suffix = generateMinSuffix(rem2, rem3, rem5, rem7, remLen);
                    sb.append(suffix);
                    return sb.toString();
                }
            }
        }

        // Шаг 3: Если в текущей длине увеличить нельзя, создаем число большей длины
        int rem2 = Math.max(0, req2);
        int rem3 = Math.max(0, req3);
        int rem5 = Math.max(0, req5);
        int rem7 = Math.max(0, req7);
        
        int targetLen = Math.max(n + 1, getMinDigitsLen(rem2, rem3, rem5, rem7));
        return generateMinSuffix(rem2, rem3, rem5, rem7, targetLen);
    }

    private int getFactor(char c, int p) {
        if (c == '0') return 0;
        int val = c - '0';
        int count = 0;
        while (val % p == 0) {
            count++;
            val /= p;
        }
        return count;
    }

    // ИСПРАВЛЕНО: Длина теперь считается строго на основе сгенерированной строки двоек/троек
    private int getMinDigitsLen(int r2, int r3, int r5, int r7) {
        return r7 + r5 + getBest23(r2, r3).length();
    }

    // ИСПРАВЛЕНО: Полностью переписанная сборка гарантирует оптимальный выбор цифр
    private String generateMinSuffix(int r2, int r3, int r5, int r7, int len) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < r7; i++) sb.append('7');
        for (int i = 0; i < r5; i++) sb.append('5');
        sb.append(getBest23(r2, r3));
        
        // Если остались свободные места, забиваем их единицами
        while (sb.length() < len) {
            sb.append('1');
        }
        
        char[] chars = sb.toString().toCharArray();
        Arrays.sort(chars); // Сортировка делает число минимальным
        return new String(chars);
    }

    // Вспомогательный метод для точного и оптимального разложения степеней двойки и тройки
    private String getBest23(int r2, int r3) {
        StringBuilder sb = new StringBuilder();
        int n9 = r3 / 2;
        r3 %= 2;
        
        // Если нечетных троек нет, просто упаковываем двойки в 8, 4, 2
        if (r3 == 0) {
            int n8 = r2 / 3;
            int rem2 = r2 % 3;
            for (int i = 0; i < n8; i++) sb.append('8');
            if (rem2 == 1) sb.append('2');
            else if (rem2 == 2) sb.append('4');
            for (int i = 0; i < n9; i++) sb.append('9');
            char[] arr = sb.toString().toCharArray();
            Arrays.sort(arr);
            return new String(arr);
        }
        
        // Если осталась одна тройка (r3 == 1), у нас есть два пути:
        
        // Путь А: Использовать шестерку '6' (забирает одну 3 и одну 2)
        StringBuilder sbA = new StringBuilder();
        int r2A = Math.max(0, r2 - 1);
        int n8A = r2A / 3;
        int rem2A = r2A % 3;
        sbA.append('6');
        for (int i = 0; i < n8A; i++) sbA.append('8');
        if (rem2A == 1) sbA.append('2');
        else if (rem2A == 2) sbA.append('4');
        for (int i = 0; i < n9; i++) sbA.append('9');
        char[] arrA = sbA.toString().toCharArray();
        Arrays.sort(arrA);
        String strA = new String(arrA);
        
        // Путь Б: Использовать тройку '3' (двойки собираются независимо)
        StringBuilder sbB = new StringBuilder();
        int n8B = r2 / 3;
        int rem2B = r2 % 3;
        sbB.append('3');
        for (int i = 0; i < n8B; i++) sbB.append('8');
        if (rem2B == 1) sbB.append('2');
        else if (rem2B == 2) sbB.append('4');
        for (int i = 0; i < n9; i++) sbB.append('9');
        char[] arrB = sbB.toString().toCharArray();
        Arrays.sort(arrB);
        String strB = new String(arrB);
        
        // Выбираем строчку покороче, а при равной длине — лексикографически меньшую
        if (strA.length() < strB.length()) return strA;
        if (strB.length() < strA.length()) return strB;
        return strA.compareTo(strB) < 0 ? strA : strB;
    }
}
