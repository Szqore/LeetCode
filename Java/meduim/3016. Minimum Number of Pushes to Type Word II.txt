import java.util.Arrays;

public class Solution {
    public int minimumPushes(String word) {
        // Step 1: Подсчет частоты каждого символа
        int[] counts = new int[26];
        for (int i = 0; i < word.length(); i++) {
            counts[word.charAt(i) - 'a']++;
        }
        
        // Step 2: Сортировка частот по возрастанию
        Arrays.sort(counts);
        
        int totalPushes = 0;
        int distinctLetters = 0;
        
        // Step 3: Распределение букв начиная с самых частых (с конца массива)
        for (int i = 25; i >= 0; i--) {
            if (counts[i] == 0) break; // Если буквы закончились, выходим
            
            // Стоимость нажатия увеличивается на 1 через каждые 8 уникальных букв
            int pressCost = (distinctLetters / 8) + 1;
            totalPushes += counts[i] * pressCost;
            
            distinctLetters++;
        }
        
        return totalPushes;
    }
}
