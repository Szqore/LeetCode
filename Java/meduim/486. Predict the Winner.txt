public class Solution { //task 486
    public boolean predictTheWinner(int[] nums) {
        int n = nums.length;
        // Если количество элементов четное, Первый игрок всегда может победить
        if (n % 2 == 0) {
            return true;
        }
        
        int[] dp = new int[n];
        
        // Базовый случай: когда остался один элемент, игрок забирает его стоимость
        for (int i = 0; i < n; i++) {
            dp[i] = nums[i];
        }
        
        // Пошагово увеличиваем длину подмассива
        for (int diff = 1; diff < n; diff++) {
            for (int i = 0; i < n - diff; i++) {
                int j = i + diff;
                // Текущий игрок выбирает максимум между взятием левого (i) или правого (j) элемента
                dp[i] = Math.max(nums[i] - dp[i + 1], nums[j] - dp[i]);
            }
        }
        
        // Если разница очков >= 0, Первый игрок победил
        return dp[0] >= 0;
    }
}
