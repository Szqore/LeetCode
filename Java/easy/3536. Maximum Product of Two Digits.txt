class Solution { // task 3536
    public int maxProduct(int n) {
        // Инициализируем переменные для хранения двух наибольших цифр
        int max1 = 0; // Самая большая цифра
        int max2 = 0; // Вторая по величине цифра

        // Цикл работает, пока в числе n остались неотсортированные цифры
        // После каждой итерации отбрасываем последнюю цифру (деление на 10)
        for (; n > 0; n /= 10) {
            // Извлекаем последнюю цифру числа n
            int currentDigit = n % 10;

            // Если текущая цифра строго больше самого большого максимума
            if (max1 < currentDigit) {
                max2 = max1;       // Старый первый максимум становится вторым
                max1 = currentDigit; // Записываем новый абсолютный максимум
            } 
            // Иначе, если цифра не больше max1, но строго больше второго максимума
            else if (max2 < currentDigit) {
                max2 = currentDigit; // Обновляем только второй максимум
            }
        }

        // Возвращаем произведение двух найденных наибольших цифр
        return max1 * max2;
    }
}
